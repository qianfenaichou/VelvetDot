import re, os
path = 'stage3/3.3/sections/methodology.md'
text = open(path, encoding='utf-8').read()
lines = text.split('\n')

CAND_DIRS = [
    '.', 'design', 'design/verify/cases', 'design/verify/perf', 'design/verify/area',
    'repos/XiangShan/src/main/scala/xiangshan',
    'repos/XiangShan/src/main/scala/xiangshan/backend/fu',
    'repos/XiangShan/src/main/scala/xiangshan/backend/fu/wrapper',
    'repos/XiangShan/src/main/scala/xiangshan/backend/fu/NewCSR',
    'repos/XiangShan/src/main/scala/xiangshan/backend/rob',
    'repos/XiangShan/yunsuan/src/main/scala/yunsuan/vector/vectorIMAC',
    'stage3/3.3','stage3/3.3/evidence', 'stage2/2.2-rtl/wave',
]
EXT = {'agner.txt': '/tmp/agner.txt', 'kalibera.txt': '/tmp/kalibera.txt', 'nanhu.txt': '/tmp/nanhu.txt',
       'synth.tcl': '/tmp/orfs_synth.tcl', 'load.tcl': '/tmp/orfs_load.tcl'}

def resolve(f):
    if os.path.exists(f): return f
    if f in EXT and os.path.exists(EXT[f]): return EXT[f]
    base = os.path.basename(f)
    for d in CAND_DIRS:
        p = os.path.join(d, base)
        if os.path.exists(p): return p
    if base in EXT and os.path.exists(EXT[base]): return EXT[base]
    return None

pat = re.compile(r'`([A-Za-z0-9_./\-]+\.[A-Za-z0-9]+):([0-9][0-9,\-]*)`')
bad, n, seen = [], 0, set()
for ln, line in enumerate(lines, 1):
    for m in pat.finditer(line):
        f, spec = m.group(1), m.group(2)
        if (f, spec) in seen: continue
        seen.add((f, spec)); n += 1
        real = resolve(f)
        if real is None:
            bad.append((ln, f, spec, 'FILE_MISSING')); continue
        total = sum(1 for _ in open(real, encoding='utf-8', errors='replace'))
        for part in spec.split(','):
            part = part.strip()
            if not part: continue
            if '-' in part:
                a, b = part.split('-', 1)
                try: a, b = int(a), int(b)
                except ValueError: bad.append((ln, f, spec, 'BAD_RANGE')); continue
                if a < 1 or b > total or a > b:
                    bad.append((ln, f, spec, f'RANGE_OUT {a}-{b} total={total}'))
            else:
                try: v = int(part)
                except ValueError: bad.append((ln, f, spec, 'BAD_NUM')); continue
                if v < 1 or v > total: bad.append((ln, f, spec, f'LINE_OUT {v} total={total}'))
print(f'refs={n} BAD={len(bad)}')
for b in bad: print('  ', b)
